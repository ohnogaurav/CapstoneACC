Example Queries

View Sample Data SELECT * FROM crimes LIMIT 10;
Top 10 Crime Types SELECT primary_type, COUNT(*) as count FROM crimes GROUP BY primary_type ORDER BY count DESC LIMIT 10;
Crimes by Year SELECT Year, COUNT(*) as total_crimes FROM crimes WHERE Year IS NOT NULL GROUP BY Year ORDER BY Year;
Arrests by Crime Type SELECT primary_type, SUM(CASE WHEN arrest = 1 OR arrest = \"True\" THEN 1 ELSE 0 END) as arrests FROM crimes GROUP BY primary_type ORDER BY arrests DESC LIMIT 5;
Crimes by Community Area SELECT community_area, COUNT(*) as crime_count FROM crimes WHERE community_area IS NOT NULL GROUP BY community_area ORDER BY crime_count DESC LIMIT 10;
